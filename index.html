async function convertPhoto(file){ if(!file) throw Error("Nu ai ales niciun fișier."); if(!file.type.startsWith("image/")) throw Error("Fișierul nu este recunoscut ca fotografie."); const url=URL.createObjectURL(file); try{ const img=new Image(); await new Promise((resolve,reject)=>{ img.onload=resolve; img.onerror=()=>reject(Error("Telefonul nu poate deschide fotografia. Încearcă format JPG.")); img.src=url; }); const width=img.naturalWidth; const height=img.naturalHeight; if(!width||!height) throw Error("Fotografie invalidă."); const scale=Math.min(1,1200/Math.max(width,height)); const canvas=document.createElement("canvas"); canvas.width=Math.round(width*scale); canvas.height=Math.round(height*scale); const ctx=canvas.getContext("2d"); if(!ctx) throw Error("Eroare la pregătirea fotografiei."); ctx.drawImage(img,0,0,canvas.width,canvas.height); const blob=await new Promise((resolve,reject)=>{ canvas.toBlob(result=>{ if(result) resolve(result); else reject(Error("Conversia fotografiei a eșuat.")); },"image/jpeg",0.8); }); canvas.width=1; canvas.height=1; return await readBlob(blob); }finally{ URL.revokeObjectURL(url); } }
